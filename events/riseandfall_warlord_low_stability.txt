namespace = warlord

# Liege-facing notification: informs the ruler that warlords are rising among vassals
warlord.1001 = {
    type = character_event
    title = warlord.1001.t
    desc = warlord.1001.d
    theme = realm

    left_portrait = root

    option = {
        name = warlord.1001.a
        ai_chance = { base = 100 }
        # Grant the liege a realm law and run the hidden_effect so the iteration runs in effect scope
        if = {
            limit = { NOT = { has_realm_law = crown_authority_0 } }
            add_realm_law = crown_authority_0
        }
        hidden_effect = {
            riseandfall_warlord_notify_vassals = yes
            # The crisis is the only window in which new vassals can join.
            add_character_flag = { flag = rf_warlord_crisis_active years = 5 }
            # Set the title-based cooldown so it survives succession.
            riseandfall_set_low_stability_cooldown_se = yes
        }
    }
}

# Vassal-facing decision: vassals may choose to become a warlord or remain loyal
warlord.1002 = {
    type = character_event
    title = warlord.1002.t
    desc = warlord.1002.d
    theme = realm

    left_portrait = liege
    right_portrait = root

    option = {
        name = warlord.1002.a
        # Become a regional Warlord and join the internal power struggle.
        riseandfall_warlord_become_warlord = yes
        ai_chance = {
            # A Warlord crisis is a broad uprising; most AI vassals join unless
            # their personality or loyalty gives them a clear reason not to.
            base = 120
            modifier = { has_trait = ambitious add = 25 }
            modifier = { has_trait = disloyal add = 25 }
            modifier = { has_trait = content add = -25 }
            modifier = { has_trait = craven add = -20 }
            modifier = { opinion = { target = liege value <= -25 } add = 25 }
            modifier = { opinion = { target = liege value >= 25 } add = -20 }
        }
    }

    option = {
        name = warlord.1002.b
        # Remain loyal: prevent repeat offers during this and the next crisis.
        # For the player, also set the loyal flag so the path manager decision becomes available
        if = {
            limit = { is_ai = no }
            add_character_flag = riseandfall_warlord_player_loyal
        }
        riseandfall_warlord_remain_loyal = yes
        ai_chance = { base = 20 }
    }
}

# Legacy player-facing Warlord choice. AI continuation and retirement are
# calculated directly on the yearly pulse so percentage chances are exact.
warlord.1003 = {
    type = character_event
    title = warlord.1003.t
    desc = warlord.1003.d
    theme = realm

    left_portrait = liege
    right_portrait = root

    option = {
        name = warlord.1003.a
        # Continue as a regional Warlord.
        ai_chance = { base = 100 }
    }
    option = {
        name = warlord.1003.b
        # Abandon the Warlord path.
        riseandfall_warlord_abandon_path = yes
        ai_chance = { base = 0 }
    }
}
