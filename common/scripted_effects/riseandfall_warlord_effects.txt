# Cleanup effect: enforce only basic Warlord trait consistency.
riseandfall_warlord_cleanup_se = {
    every_ruler = {
        limit = { is_landed = no has_trait = warlord }
        remove_trait = warlord
    }

    # Independent Warlords cease to be part of a liege's internal crisis.
    # Do not add retiredwarlord: compatibility mods own that trait's lifecycle.
    every_independent_ruler = {
        limit = { has_trait = warlord }
        remove_trait = warlord
    }
}

# Called from on_imprison when a liege imprisons one of their Warlord vassals.
riseandfall_warlord_imprisoned_se = {
    if = {
        limit = {
            has_trait = warlord
            target_is_liege_or_above = scope:imprisoner
        }
        riseandfall_warlord_abandon_path = yes
    }
}

# Expected scope: liege with Warlord vassals. Legitimacy loss scales with the
# share of direct landed vassals participating in the Warlord movement.
riseandfall_warlord_legitimacy_loss_se = {
    if = {
        limit = {
            has_legitimacy = yes
            any_vassal = {
                primary_title.tier >= tier_county
                has_trait = warlord
            }
        }
        set_variable = { name = rf_total_vassals value = 0 }
        set_variable = { name = rf_warlord_vassals value = 0 }
        every_vassal = {
            limit = { primary_title.tier >= tier_county }
            prev = { change_variable = { name = rf_total_vassals add = 1 } }
            if = {
                limit = { has_trait = warlord }
                prev = { change_variable = { name = rf_warlord_vassals add = 1 } }
            }
        }
        if = {
            limit = { var:rf_total_vassals > 0 }
            set_variable = { name = rf_legitimacy_loss value = var:rf_warlord_vassals }
            change_variable = { name = rf_legitimacy_loss multiply = -100 }
            change_variable = { name = rf_legitimacy_loss divide = var:rf_total_vassals }
            if = {
                limit = { var:rf_legitimacy_loss < 0 }
                add_legitimacy = var:rf_legitimacy_loss
            }
            remove_variable = rf_legitimacy_loss
        }
        remove_variable = rf_total_vassals
        remove_variable = rf_warlord_vassals
    }
}
