# Quarterly fallback: re-sync puppet master court position for any liege whose position
# is missing while a puppet_regency is still active. Ensures refill within 3 months
# if the 1-day revoke event somehow fails.
quarterly_playable_pulse = {
    on_actions = {
        riseandfall_puppet_master_quarterly_resync
    }
}

riseandfall_puppet_master_quarterly_resync = {
    trigger = {
        has_game_rule = riseandfall_low_stability_events_enabled
        has_diarchy_type = puppet_regency
        has_active_diarchy = yes
        NOT = { employs_court_position = riseandfall_puppet_master_court_position }
    }
    effect = {
        riseandfall_sync_puppet_master_court_position = yes
    }
}

riseandfall_low_stability_roll = {
    trigger = {
        has_game_rule = riseandfall_low_stability_events_enabled
        is_landed = yes
        is_independent_ruler = yes
        # Low-stability incidents are reserved for duchy-tier realms and above.
        primary_title.tier >= tier_duchy

riseandfall_mark_puppet_succession_on_death = {
    trigger = {
        has_game_rule = riseandfall_low_stability_events_enabled
        is_independent_ruler = yes
        has_diarchy_type = puppet_regency
        has_active_diarchy = yes
    }
    effect = {
        # Death fires before inheritance has fully settled; mark the likely heir
        # and resolve the puppet succession after title inheritance.
        clear_saved_scope = rf_dying_puppet_master
        if = {
            limit = { exists = diarch }
            diarch = { save_scope_as = rf_dying_puppet_master }
        }
        if = {
            limit = {
                exists = primary_heir
                primary_heir = { is_alive = yes }
            }
            primary_heir = {
                riseandfall_clear_puppet_succession_pending_se = yes
                add_character_flag = { flag = rf_puppet_succession_pending days = 30 }
                set_variable = { name = rf_puppet_succession_predecessor value = root }
                if = {
                    limit = { exists = scope:rf_dying_puppet_master }
                    set_variable = { name = rf_puppet_succession_diarch value = scope:rf_dying_puppet_master }
                }
                if = {
                    limit = { root = { has_variable = riseandfall_puppet_regency_years } }
                    set_variable = { name = rf_puppet_succession_years value = root.var:riseandfall_puppet_regency_years }
                }
                if = { limit = { is_landed = yes } add_character_flag = { flag = rf_puppet_succession_heir_was_landed days = 30 } }
                if = { limit = { root = { government_allows = administrative } } add_character_flag = { flag = rf_puppet_succession_predecessor_admin days = 30 } }
            }
        }
        else_if = {
            limit = {
                exists = player_heir
                player_heir = { is_alive = yes }
            }
            player_heir = {
                riseandfall_clear_puppet_succession_pending_se = yes
                add_character_flag = { flag = rf_puppet_succession_pending days = 30 }
                set_variable = { name = rf_puppet_succession_predecessor value = root }
                if = {
                    limit = { exists = scope:rf_dying_puppet_master }
                    set_variable = { name = rf_puppet_succession_diarch value = scope:rf_dying_puppet_master }
                }
                if = {
                    limit = { root = { has_variable = riseandfall_puppet_regency_years } }
                    set_variable = { name = rf_puppet_succession_years value = root.var:riseandfall_puppet_regency_years }
                }
                if = { limit = { is_landed = yes } add_character_flag = { flag = rf_puppet_succession_heir_was_landed days = 30 } }
                if = { limit = { root = { government_allows = administrative } } add_character_flag = { flag = rf_puppet_succession_predecessor_admin days = 30 } }
            }
        }
        clear_saved_scope = rf_dying_puppet_master
    }
}

riseandfall_resolve_puppet_succession_on_inheritance = {
    trigger = {
        has_game_rule = riseandfall_low_stability_events_enabled
        has_character_flag = rf_puppet_succession_pending
        has_variable = rf_puppet_succession_predecessor
        exists = scope:previous_holder
        var:rf_puppet_succession_predecessor ?= scope:previous_holder
        is_landed = yes
        is_independent_ruler = yes
        NOT = { has_character_flag = rf_puppet_succession_event_queued }
    }
    effect = {
        if = {
            limit = {
                has_active_diarchy = yes
                NOT = { has_diarchy_type = puppet_regency }
            }
            riseandfall_clear_puppet_succession_pending_se = yes
        }
        else_if = {
            limit = {
                OR = {
                    has_character_flag = rf_puppet_succession_heir_was_landed
                    has_character_flag = rf_puppet_succession_predecessor_admin
                    government_allows = administrative
                }
            }
            add_character_flag = { flag = rf_puppet_succession_event_queued days = 30 }
            trigger_event = { id = riseandfall_puppet_master.110 delayed = yes }
        }
        else = {
            riseandfall_accept_puppet_succession_status_quo_se = yes
            riseandfall_clear_puppet_succession_pending_se = yes
        }
    }
}


riseandfall_warlord = {
    trigger = {
        has_game_rule = riseandfall_low_stability_events_enabled
    }
    effect = {
        # Cleanup: handle unlanded/independent warlords
        riseandfall_warlord_cleanup_se = yes

        # Evaluate each independent realm once. The count and territory share
        # cover direct Warlords and their full subrealms.
        every_independent_ruler = {
            limit = {
                is_landed = yes
            }
            riseandfall_warlord_refresh_crisis_strength_se = yes
            every_vassal = {
                limit = {
                    is_ai = yes
                    is_landed = yes
                    has_trait = warlord
                }
                riseandfall_warlord_ai_reconsider_se = yes
            }
            # Only an active Warlord crisis attracts new followers. AI recruits
            # are silent; the initial event is the player-facing invitation.
            if = {
                limit = { has_character_flag = rf_warlord_crisis_active }
                every_vassal = {
                    limit = {
                        is_ai = yes
                        is_landed = yes
                        NOT = { has_trait = warlord }
                        NOT = { has_variable = riseandfall_warlord_choice_made }
                        NOT = { has_variable = riseandfall_warlord_retired_cooldown }
                    }
                    riseandfall_warlord_ai_join_crisis_se = yes
                }
            }
        }
    }
}

# On_actions for puppet masters (diarchs): AI-only unlanded diarchs should be given spouses by their liege's court.
# We iterate each ruler with a puppet_regency diarchy and pick any courtier who is that ruler's diarch, then call the marriage scripted effects.
riseandfall_marry_male_puppet_master_target = {
    trigger = {
        has_game_rule = riseandfall_low_stability_events_enabled
    }
    effect = {
        every_ruler = {
            limit = { has_diarchy_type = puppet_regency }
            save_scope_as = liege
            every_courtier = {
                limit = {
                    is_diarch_of_target = scope:liege
                    is_ai = yes
                    is_landed = no
                    is_ruler = no
                    NOT = { has_trait_with_flag = can_not_marry }
                    is_male = yes
                    is_adult = yes
                    is_married = no
                    is_betrothed = no
                }
                save_scope_as = diarch_candidate
                # call the scripted effect on the diarch character (root)
                riseandfall_marry_puppet_master_male_se = yes
            }
        }
    }
}
