    desc = riseandfall_grant_duchies_desc
    selection_tooltip = riseandfall_grant_duchies_tooltip

riseandfall_grant_counties = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
    sort_order = 20
    ai_check_interval = 12
    desc = riseandfall_grant_counties_desc
    selection_tooltip = riseandfall_grant_counties_desc
    decision_group_type = riseandfall

    # County rulers cannot safely create a subordinate county vassal.
    is_shown = {
        is_landed = yes
        primary_title.tier >= tier_duchy
        NOT = { exists = global_var:riseandfall_hide_other_decisions }
    }

    is_valid = { is_landed = yes primary_title.tier >= tier_duchy }

    # Manual grant: run the counties grant scripted effect immediately
    effect = { riseandfall_grant_counties_se = yes }

    # AI: only consider this decision if AI and they're over the domain limit
    # domain_limit_available is negative when above the limit; this AI check wants strongly over the limit
    ai_potential = { domain_limit_available < -20 }

    # Make AI very likely to pick this when ai_potential is true
    ai_will_do = { base = 100 }
}

# Player: toggle guardian assignment for house children in the ruler's court.
riseandfall_automate_house_child_guardians = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
    sort_order = 24
    ai_check_interval = 0
    desc = riseandfall_automate_house_child_guardians_desc
    selection_tooltip = riseandfall_automate_house_child_guardians_tooltip
    decision_group_type = courtier_automation

    is_shown = {
        is_landed = yes
        NOT = { exists = global_var:riseandfall_hide_other_decisions }
        exists = house
        NOT = { has_character_flag = riseandfall_auto_house_child_guardians_enabled }
    }

    is_valid = { is_landed = yes }

    effect = {
        add_character_flag = { flag = riseandfall_auto_house_child_guardians_enabled }
        # Give the player immediate feedback; future passes happen yearly.
        riseandfall_player_automate_house_child_guardians_se = yes
    }

    ai_potential = {}
    ai_will_do = { base = 0 }
}
