# Rise and Fall - Coalition low-stability triggers

# True while this ruler is the defender in an already active Coalition war.
riseandfall_coalition_target_has_active_war_trigger = {
    any_character_war = {
        using_cb = riseandfall_coalition_cb
        is_defender = prev
    }
}

# Realm Stability stores the ruler's rounded share of world realm size as a
# negative component: controlling 5% of the world produces -5. Coalition is
# reserved for rulers at that threshold or larger.
riseandfall_coalition_target_controls_five_percent_world_trigger = {
    has_variable = riseandfall_rs_realm_size_component
    NOT = { var:riseandfall_rs_realm_size_component >= -4 }
}

# This trigger is evaluated from a Coalition interaction, where scope:actor
# is the initiating neighbor and this is the proposed target.
riseandfall_coalition_target_threat_valid_trigger = {
    has_variable = rf_coalition_threat
    var:rf_coalition_threat >= 100
    riseandfall_coalition_target_controls_five_percent_world_trigger = yes
    has_variable = rf_coalition_peripheral_pct
    var:rf_coalition_peripheral_pct >= 25
    has_variable = rf_coalition_core_counties
    var:rf_coalition_core_counties > 0
    realm_size >= { value = scope:actor.realm_size multiply = 2 }
}

# Stable target requirements shared by selection, preparation, and final assembly.
riseandfall_coalition_target_core_valid_trigger = {
    is_alive = yes
    is_landed = yes
    is_independent_ruler = yes
    exists = primary_title
    riseandfall_coalition_target_controls_five_percent_world_trigger = yes
    has_variable = rf_coalition_peripheral_pct
    var:rf_coalition_peripheral_pct >= 25
    has_variable = rf_coalition_core_counties
    var:rf_coalition_core_counties > 0
    NOT = { primary_title = { has_variable = rf_coalition_postwar_cooldown } }
    NOT = { has_character_flag = rf_coalition_under_attack }
    NOT = { riseandfall_coalition_target_has_active_war_trigger = yes }
}

riseandfall_coalition_interaction_target_valid_trigger = {
    riseandfall_coalition_target_core_valid_trigger = yes
    NOT = { has_character_flag = rf_coalition_formation_active }
    NOT = { primary_title = { has_variable = rf_coalition_interaction_cooldown } }
}

# The dynamic cause is checked before Coalition is selected from the
# low-stability pool. Once that incident fires, do not recheck transient
# percentage/cooldown values during invitation assembly: only a genuinely
# invalid target may stop the already-started Coalition.
riseandfall_coalition_target_still_valid_trigger = {
    is_alive = yes
    is_landed = yes
    is_independent_ruler = yes
    exists = primary_title
    NOT = { has_character_flag = rf_coalition_under_attack }
    NOT = { riseandfall_coalition_target_has_active_war_trigger = yes }
}

# A ruler who could give the Coalition at least one legal attacker slot.
riseandfall_coalition_potential_invitee_trigger = {
    # This trigger runs inside an ordered ruler iterator, where root is not
    # retained. The formation target is saved explicitly by the caller.
    NOT = { this = scope:coalition_target }
    is_alive = yes
    is_landed = yes
    is_independent_ruler = yes
    exists = primary_title
    in_diplomatic_range = scope:coalition_target
    is_at_war = no
    NOT = { is_allied_to = scope:coalition_target }
    NOT = { has_truce = scope:coalition_target }
    NOT = { has_variable = rf_coalition_target }
    NOR = {
        government_has_flag = government_is_landless_adventurer
    }
}

# Selector-facing availability. The peripheral percentage is calculated by the
# yearly incident effect from counties, never vassal primary titles.
# Invitation-time legality. The target is saved on the invitee so tooltip and
# event option evaluation never dereference a missing transient scope.
riseandfall_coalition_invitee_can_join_trigger = {
    has_variable = rf_coalition_target
    var:rf_coalition_target ?= scope:coalition_target
    is_alive = yes
    is_landed = yes
    is_independent_ruler = yes
    exists = primary_title
    is_at_war = no
    NOT = { is_allied_to = scope:coalition_target }
    NOT = { has_truce = scope:coalition_target }
    scope:coalition_target = {
        riseandfall_coalition_target_still_valid_trigger = yes
        has_character_flag = rf_coalition_formation_active
    }
}

# Direct vassals may join the external Coalition as internal rebels. They must
# still serve the target when the invitation resolves, and existing wars or a
# personal alliance with their liege make rebellion unavailable.
riseandfall_coalition_vassal_can_join_trigger = {
    has_variable = rf_coalition_target
    var:rf_coalition_target ?= scope:coalition_target
    is_alive = yes
    is_landed = yes
    is_independent_ruler = no
    exists = primary_title
    highest_held_title_tier >= tier_county
    is_at_war = no
    NOT = { is_allied_to = scope:coalition_target }
    # This is the engine's direct-vassal check. It excludes the target's
    # subvassals and every ruler belonging to another realm.
    is_vassal_of = scope:coalition_target
    # Only vassals with actual peripheral land can use the Coalition to break
    # away. A core primary title cannot hide their outer counties.
    any_held_title = {
        tier = tier_county
        NOT = { target_is_de_jure_liege_or_above = scope:coalition_target.var:rf_coalition_core_title }
    }
    NOR = {
        government_has_flag = government_is_landless_adventurer
    }
    scope:coalition_target = {
        riseandfall_coalition_target_still_valid_trigger = yes
        has_character_flag = rf_coalition_formation_active
    }
}
